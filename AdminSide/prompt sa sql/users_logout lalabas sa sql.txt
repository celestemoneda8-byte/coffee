-- Run this SQL in your database (phpMyAdmin / MySQL CLI) to create the user_logouts table
CREATE TABLE IF NOT EXISTS user_logouts (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT DEFAULT NULL,
  username VARCHAR(150) DEFAULT NULL,
  ip_address VARCHAR(45) DEFAULT NULL,
  user_agent TEXT DEFAULT NULL,
  logged_out_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Optional: add last_logout column to users table (if you want to track it)
-- Only run if your users table exists and doesn't already have last_logout
ALTER TABLE users
  ADD COLUMN IF NOT EXISTS last_logout TIMESTAMP NULL DEFAULT NULL;
